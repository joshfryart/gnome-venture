-- ROLLOUT frames (the other 15): sitting family reuses sit(), the five action poses are built like pounce/nap ------------
local function limb(c, info, pts, r0, r1, o)
  local m = L.tube(pts, function(t) return r0 + (r1 - r0) * t end)
  c:part(m, o or { ws = 1, wl = 2 })
  info.legs = info.legs:copy():add(m)
  return m
end
local function paw(c, info, cx, cy, rx, ry, o)
  local m = L.ellipse(cx, cy, rx, ry)
  c:part(m, o or { ws = 1, wl = 1, noao = true })
  info.paws = info.paws:copy():add(m)
  return m
end
local function raisedPaw(c, info, cx, cy, pads)
  paw(c, info, cx, cy, 3.3, 2.6)
  if pads then
    for _, p in ipairs({ { 0, 1, 'P' }, { 1, 1, 'P' }, { -1, -1, 'p' }, { 0, -2, 'p' }, { 1, -2, 'p' }, { 2, -1, 'p' } }) do
      c:put(floor(cx) + p[1], floor(cy) + p[2], p[3], "detail")
    end
  else
    toes(c, { floor(cx) - 1, floor(cx) + 1 }, floor(cy) - 1, floor(cy))
  end
end

local function crouch(o, frame, ox, oy, eyes, build, tailPts, tailTip, gy)
  local c, info = L.canvas(64, 64), { frame = frame, ox = ox, oy = oy }
  info.legs, info.paws = L.mask(), L.mask()
  shadow(c, 30, 58, 22, 2, 15)
  if o.tail == "bobtail" then
    local m = L.ellipse(tailPts[1][1], tailPts[1][2] - 2, 2.8, 2.8) c:part(m, { ws = 1, wl = 1 }) info.tail = m info.tailT = {}
  else
    local m, T = L.tube(tailPts, BRUSH)
    c:part(m, { ws = 1, wl = 2 })
    info.tail, info.tailT, info.tailPts = m, T, tailPts
    tailTip(c, tailTip)
  end
  build(c, info)
  head(c, info, ox, oy, eyes, o.ears, false, nil)
  if info.after then info.after(c) end
  finish(c, info, gy)
  return c, info
end

CATS.jump = function(o)
  local c, info = L.canvas(64, 64), { frame = "jump", ox = 7, oy = 2 }
  info.legs, info.paws = L.mask(), L.mask()
  shadow(c, 32, 59.5, 14, 1.5, 9)
  local tp = { { 38, 50 }, { 45, 53 }, { 51, 51 }, { 55, 46 }, { 56, 42 } }
  local m, T = L.tube(tp, BRUSH) c:part(m, { ws = 1, wl = 2 })
  info.tail, info.tailT, info.tailPts = m, T, tp
  tailTip(c, { { 56, 43, 'g' }, { 55, 44, 'L' } })
  limb(c, info, { { 26, 49 }, { 24, 53 }, { 24, 56 } }, 2.6, 2.1)
  limb(c, info, { { 38, 49 }, { 40, 53 }, { 40, 56 } }, 2.6, 2.1)
  paw(c, info, 24, 56, 3.6, 1.9) paw(c, info, 40, 56, 3.6, 1.9)
  toes(c, { 23, 25, 39, 41 }, 56, 57)
  local chest = L.ellipse(32, 44, 9.6, 8.2)
  c:part(chest, { ws = 2, wl = 3 })
  L.ellipse(32, 45, 4, 5):each(function(x, y) local t = c:get(x, y) if t and L.isCoat(t) then c:put(x, y, L.step(t, 1), "light") end end)
  info.chest, info.haunch = chest, chest
  info.hcx, info.hcy, info.hrx, info.hry = 32, 44, 9.6, 8.2
  for _, s in ipairs({ { 24, 12, 7, 6 }, { 40, 52, 57, 58 } }) do
    limb(c, info, { { s[1], 38 }, { s[2], 33 }, { s[3], 23 }, { s[4], 15 } }, 2.5, 2.0)
    raisedPaw(c, info, s[4] + 0.5, 12.5, false)
  end
  head(c, info, 7, 2, "open", o.ears, false, nil)
  finish(c, info, nil)
  return c, info
end
